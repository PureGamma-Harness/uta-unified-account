CREATE TABLE main.auth_config (
 singleton INTEGER PRIMARY KEY CHECK(singleton=1),
 format INTEGER NOT NULL CHECK(typeof(format)='integer' AND format=2),
 canonical_path TEXT NOT NULL, auth_authority_id TEXT NOT NULL,
 tenant TEXT NOT NULL, audience TEXT NOT NULL, producer_binding TEXT NOT NULL,
 domain_id TEXT NOT NULL, domain_epoch INTEGER NOT NULL CHECK(typeof(domain_epoch)='integer' AND domain_epoch=1),
 account_id TEXT NOT NULL,
 past_window INTEGER NOT NULL CHECK(typeof(past_window)='integer' AND past_window BETWEEN 1 AND 300000),
 future_skew INTEGER NOT NULL CHECK(typeof(future_skew)='integer' AND future_skew BETWEEN 0 AND 30000),
 max_nonces INTEGER NOT NULL CHECK(typeof(max_nonces)='integer' AND max_nonces BETWEEN 2 AND 1000000),
 query_nonce_reserve INTEGER NOT NULL CHECK(typeof(query_nonce_reserve)='integer' AND query_nonce_reserve>0 AND query_nonce_reserve<max_nonces),
 max_audit INTEGER NOT NULL CHECK(typeof(max_audit)='integer' AND max_audit BETWEEN 3 AND 1000000),
 query_audit_reserve INTEGER NOT NULL CHECK(typeof(query_audit_reserve)='integer' AND query_audit_reserve>0),
 control_audit_reserve INTEGER NOT NULL CHECK(typeof(control_audit_reserve)='integer' AND control_audit_reserve>0),
 max_acceptances INTEGER NOT NULL CHECK(typeof(max_acceptances)='integer' AND max_acceptances BETWEEN 1 AND 10000),
 max_grants INTEGER NOT NULL CHECK(typeof(max_grants)='integer' AND max_grants BETWEEN 1 AND 10000),
 highwater INTEGER NOT NULL CHECK(typeof(highwater)='integer' AND highwater BETWEEN 0 AND 9223372036854775807),
 revision INTEGER NOT NULL CHECK(typeof(revision)='integer' AND revision BETWEEN 0 AND 9223372036854775807),
 grant_revision INTEGER NOT NULL CHECK(typeof(grant_revision)='integer' AND grant_revision BETWEEN 0 AND 9223372036854775807),
 fingerprint TEXT NOT NULL,
 CHECK(query_audit_reserve+control_audit_reserve<max_audit)
);
CREATE TABLE main.auth_keys (
 key_id TEXT PRIMARY KEY NOT NULL, identity TEXT NOT NULL,
 role TEXT NOT NULL CHECK(role IN ('admin','reader','funding','risk','trader')),
 account_id TEXT,
 not_before_ms INTEGER NOT NULL CHECK(typeof(not_before_ms)='integer' AND not_before_ms BETWEEN 0 AND 9223372036854775806),
 expires_at_ms INTEGER NOT NULL CHECK(typeof(expires_at_ms)='integer' AND expires_at_ms>not_before_ms AND expires_at_ms<=9223372036854775807),
 revoked INTEGER NOT NULL CHECK(typeof(revoked)='integer' AND revoked IN (0,1)),
 UNIQUE(key_id,identity),
 CHECK((role IN ('reader','funding','trader') AND account_id IS NOT NULL) OR (role IN ('admin','risk') AND account_id IS NULL))
);
CREATE TABLE main.auth_nonces (
 key_id TEXT NOT NULL REFERENCES auth_keys(key_id), nonce_hash TEXT NOT NULL,
 expires_ms INTEGER NOT NULL CHECK(typeof(expires_ms)='integer' AND expires_ms BETWEEN 0 AND 9223372036854775807),
 PRIMARY KEY(key_id,nonce_hash)
);
CREATE INDEX main.auth_nonce_expiry ON auth_nonces(expires_ms);
CREATE TABLE main.auth_audit (
 sequence INTEGER PRIMARY KEY CHECK(typeof(sequence)='integer' AND sequence>0),
 event TEXT NOT NULL, outcome TEXT NOT NULL, request_kind TEXT NOT NULL,
 key_id TEXT, key_identity TEXT, nonce_hash TEXT, request_hash TEXT,
 request_ms INTEGER CHECK(request_ms IS NULL OR (typeof(request_ms)='integer' AND request_ms BETWEEN 0 AND 9223372036854775807)),
 server_ms INTEGER CHECK(server_ms IS NULL OR (typeof(server_ms)='integer' AND server_ms BETWEEN 0 AND 9223372036854775807)),
 revision INTEGER NOT NULL CHECK(typeof(revision)='integer' AND revision BETWEEN 0 AND 9223372036854775807),
 grant_revision INTEGER NOT NULL CHECK(typeof(grant_revision)='integer' AND grant_revision BETWEEN 0 AND 9223372036854775807),
 business_key TEXT, grant_id TEXT, original_acceptance_id TEXT,
 FOREIGN KEY(key_id,key_identity) REFERENCES auth_keys(key_id,identity),
 FOREIGN KEY(grant_id) REFERENCES auth_credit_grants(grant_id) DEFERRABLE INITIALLY DEFERRED,
 FOREIGN KEY(original_acceptance_id) REFERENCES auth_credit_acceptances(acceptance_id) DEFERRABLE INITIALLY DEFERRED
);
CREATE TABLE main.auth_credit_grants (
 grant_id TEXT PRIMARY KEY NOT NULL, key_id TEXT NOT NULL REFERENCES auth_keys(key_id),
 account_id TEXT NOT NULL, asset TEXT NOT NULL CHECK(asset='BTC'),
 operation TEXT NOT NULL CHECK(operation IN ('credit.borrow','credit.repay','credit.read')),
 grant_epoch INTEGER NOT NULL UNIQUE CHECK(typeof(grant_epoch)='integer' AND grant_epoch>0),
 audit_sequence INTEGER NOT NULL UNIQUE REFERENCES auth_audit(sequence) DEFERRABLE INITIALLY DEFERRED,
 payload TEXT NOT NULL,
 UNIQUE(grant_id,key_id,account_id,asset,operation,grant_epoch),
 UNIQUE(grant_id,key_id,account_id,grant_epoch)
);
CREATE TABLE main.auth_credit_grant_revocations (
 grant_id TEXT PRIMARY KEY NOT NULL REFERENCES auth_credit_grants(grant_id),
 revocation_revision INTEGER NOT NULL UNIQUE CHECK(typeof(revocation_revision)='integer' AND revocation_revision>0),
 audit_sequence INTEGER NOT NULL UNIQUE REFERENCES auth_audit(sequence) DEFERRABLE INITIALLY DEFERRED,
 payload TEXT NOT NULL
);
CREATE TABLE main.auth_credit_acceptances (
 acceptance_id TEXT PRIMARY KEY NOT NULL,
 domain_id TEXT NOT NULL, tenant TEXT NOT NULL, account_id TEXT NOT NULL, command_id TEXT NOT NULL,
 kind TEXT NOT NULL CHECK(kind IN ('BORROW','REPAY')),
 command_payload TEXT NOT NULL, payload_hash TEXT NOT NULL,
 auth_audit_sequence INTEGER NOT NULL UNIQUE REFERENCES auth_audit(sequence),
 key_id TEXT NOT NULL, key_identity TEXT NOT NULL, grant_id TEXT NOT NULL,
 grant_epoch INTEGER NOT NULL CHECK(typeof(grant_epoch)='integer' AND grant_epoch>0),
 profile_revision INTEGER NOT NULL CHECK(typeof(profile_revision)='integer' AND profile_revision>0),
 grant_revision INTEGER NOT NULL CHECK(typeof(grant_revision)='integer' AND grant_revision>=grant_epoch),
 domain_epoch INTEGER NOT NULL CHECK(typeof(domain_epoch)='integer' AND domain_epoch=1),
 request_hash TEXT NOT NULL, body_hash TEXT NOT NULL, nonce_hash TEXT NOT NULL,
 request_ms INTEGER NOT NULL CHECK(typeof(request_ms)='integer' AND request_ms BETWEEN 0 AND 9223372036854775807),
 auth_checked_server_ms INTEGER NOT NULL CHECK(typeof(auth_checked_server_ms)='integer' AND auth_checked_server_ms BETWEEN 0 AND 9223372036854775807),
 payload TEXT NOT NULL,
 UNIQUE(domain_id,tenant,account_id,command_id),
 FOREIGN KEY(key_id,key_identity) REFERENCES auth_keys(key_id,identity),
 FOREIGN KEY(grant_id,key_id,account_id,grant_epoch)
   REFERENCES auth_credit_grants(grant_id,key_id,account_id,grant_epoch)
);
