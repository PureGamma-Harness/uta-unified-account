def reference_frame(hex_value):
    """Classify only the literal finite review vectors; never read a socket."""
    raw=bytes.fromhex(hex_value)
    first,second=raw[:2]; fin=bool(first&128);opcode=first&15;masked=bool(second&128)
    width=second&127; index=2
    if first&112 or opcode not in (0,1,2,8,9,10) or not masked:
        return {'close':1002,'payload_allocated':False}
    if width==126:
        size=int.from_bytes(raw[index:index+2],'big');index+=2
        if size<126:return {'close':1002,'payload_allocated':False}
    elif width==127:
        extended=raw[index:index+8];index+=8
        if extended[0]&128:return {'close':1002,'payload_allocated':False}
        size=int.from_bytes(extended,'big')
        if size<65536:return {'close':1002,'payload_allocated':False}
    else:size=width
    if opcode>=8 and (not fin or size>125):return {'close':1002,'payload_allocated':False}
    if opcode==0:return {'close':1002,'payload_allocated':False}
    if opcode in (1,2):return {'close':1008 if opcode==1 else 1003,'payload_allocated':False}
    key=raw[index:index+4];index+=4
    assert len(key)==4 and len(raw)==index+size
    body=bytes(value^key[i%4] for i,value in enumerate(raw[index:]))
    if opcode==9:return {'close':None,'reply_hex':(bytes([138,len(body)])+body).hex(),'payload_hex':body.hex()}
    if opcode==10:return {'close':None,'pong_payload_hex':body.hex()}
    if len(body)==1:return {'close':1002,'payload_allocated':True}
    if not body:return {'close':None,'incoming_close':None,'reply_hex':'8800'}
    code=int.from_bytes(body[:2],'big')
    # Explicit review profile for known codes and application ranges. The
    # design must select its own complete registry policy before runtime GO.
    valid=code in (1000,1001,1002,1003,1007,1008,1009,1010,1011,1012,1013,1014) or 3000<=code<5000
    if not valid:return {'close':1002,'incoming_close':code,'payload_allocated':True}
    try:body[2:].decode('utf-8')
    except UnicodeDecodeError:return {'close':1007,'incoming_close':code,'payload_allocated':True}
    reply=1000 if code==1010 else code
    return {'close':None,'incoming_close':code,'reply_hex':(b'\x88\x02'+reply.to_bytes(2,'big')).hex()}
