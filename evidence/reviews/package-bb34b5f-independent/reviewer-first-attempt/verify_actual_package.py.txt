def fingerprint(path):
    path = Path(path)
    h = sha256()
    with path.open("rb") as stream:
        for chunk in iter(lambda: stream.read(65536), b""):
            h.update(chunk)
    result = {"bytes": path.stat().st_size, "sha256": h.hexdigest()}
    references[str(path)] = result
    return result
def read(path):
    path = Path(path)
    assert path.stat().st_size <= 4 * 1024 * 1024
    value = json.loads(path.read_bytes())
    fingerprint(path)
    return value
def git(*args):
    return subprocess.check_output(["git", *args], cwd=SOURCE, text=True).strip()
def archive_member(stream):
    h = sha256(); size = 0
    for chunk in iter(lambda: stream.read(65536), b""):
        size += len(chunk)
        assert size < 64 * 1024 * 1024
        h.update(chunk)
    return {"bytes": size, "sha256": h.hexdigest()}
def wheel_members(path):
    result = {}
    with zipfile.ZipFile(path) as archive:
        for member in archive.infolist():
            bounded()
            name = member.filename
            assert name not in result and not name.startswith("/") and ".." not in PurePosixPath(name).parts
            if not member.is_dir():
                with archive.open(member) as stream:
                    result[name] = archive_member(stream)
    return result
def tar_members(path):
    result = {}; root = None; seen = set()
    with tarfile.open(path, "r|gz") as archive:
        for member in archive:
            bounded()
            name = PurePosixPath(member.name)
            assert not name.is_absolute() and ".." not in name.parts
            assert member.name not in seen
            seen.add(member.name)
            if root is None:
                root = name.parts[0]
            assert name.parts[0] == root and (member.isdir() or member.isfile())
            if member.isfile():
                relative = str(PurePosixPath(*name.parts[1:]))
                with archive.extractfile(member) as stream:
                    result[relative] = archive_member(stream)
    return result
